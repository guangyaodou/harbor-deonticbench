# DeonticBench Harbor — adapter generation + evaluation commands
# Run all commands from the harbor root directory.
# Each adapter generates 173 tasks: sara_numeric(35) + sara_binary(30) + airline(80) + uscis-aao(28)

DEONTICBENCH_ROOT=/home/blab-029b/Desktop/workspace/gdou1/DeonticBench

# ==============================================================================
# STEP 1: Generate tasks
# ==============================================================================

export DEONTICBENCH_ROOT=/home/blab-029b/Desktop/workspace/gdou1/DeonticBench   

# Direct mode (free-form answer → /app/output/answer.txt)
cd adapters/deonticbench_direct
uv run python run_adapter.py \
    --deonticbench-root $DEONTICBENCH_ROOT \
    --output-dir ../../datasets/deonticbench-direct

# Zero-shot Prolog mode (agent writes /app/solution.pl, no examples)
cd ../deonticbench_zeroshot
uv run python run_adapter.py \
    --deonticbench-root $DEONTICBENCH_ROOT \
    --output-dir ../../datasets/deonticbench-zeroshot

# Few-shot Prolog mode (agent writes /app/solution.pl, examples at /app/examples.txt)
cd ../deonticbench_fewshot
uv run python run_adapter.py \
    --deonticbench-root $DEONTICBENCH_ROOT \
    --output-dir ../../datasets/deonticbench-fewshot

cd ../..

# Optional flags for run_adapter.py:
#   --subdomains sara_numeric sara_binary airline uscis-aao   filter to specific subdomains
#   --limit N                                                  generate only first N tasks
#   --clean                                                    delete output dir before generating
#   --task-ids sara_numeric/tax_case_10 ...                   generate specific task IDs


# ==============================================================================
# STEP 2: Run evaluations with Harbor (terminus-2 agent)

# Model name format for LiteLLM:
#   OpenRouter:  openrouter/<provider>/<model>
#   Together AI: together_ai/<provider>/<model>
