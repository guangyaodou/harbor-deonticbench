## DeonticBench Run Commands - GPT Models
##
## Reasoning effort can be customized via --ak reasoning_effort=<level>
## Valid levels: none, minimal, low, medium, high, default
## Applied to reasoning models: gpt-5.1, gpt-5.2, o3

# ── gpt-5.1-2025-11-13 (reasoning, effort=medium) ────────────────────────────

harbor run \
  --path datasets/deonticbench-direct \
  --agent terminus-2 \
  --model openai/gpt-5.1-2025-11-13 \
  --ae OPENAI_API_KEY=$OPENAI_API_KEY \
  --ak reasoning_effort=medium \
  --n-concurrent 2 \
  --jobs-dir jobs/deontic-direct

harbor run \
